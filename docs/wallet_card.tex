% ════════════════════════════════════════
\chead{Getting Started}
% ════════════════════════════════════════

\textbf{Start the server:}\\
\kbd{set -a; source .env; set +a}\\
\kbd{python canvas\_gui.py}\\[2pt]
Open \textbf{localhost:5050} in browser.\\[2pt]
Select a course from the dropdown to load all tabs.

\textbf{Required .env variables:}\\
\kbd{CANVAS\_API\_TOKEN} --- your token\\
\kbd{CANVAS\_API\_URL} --- with \texttt{/api/v1}\\
\kbd{LOCAL\_ROOT} --- local file directory

% ════════════════════════════════════════
\chead{Files}
% ════════════════════════════════════════

\csub{Two-Panel Layout}
Left: local files \quad Right: Canvas files\\[2pt]
\textbullet\ \textbf{Drag \& drop} local $\to$ Canvas folder\\
\textbullet\ \textbf{Type filters:} All, PDF, Python, Images, Docs, Data, Code\\
\textbullet\ \textbf{Navigate:} $\uparrow$~Up, Home, Desktop buttons\\
\textbullet\ Double-click folder to enter it\\
\textbullet\ Click path display to type a path\\
\textbullet\ Panels are resizable (drag divider)

% ════════════════════════════════════════
\chead{Assignments}
% ════════════════════════════════════════

\csub{Manual}
\textbf{+ New} $\to$ fill form $\to$ Create\\
Click sidebar item $\to$ edit in split editor\\
\textbf{Save Changes} appears on any edit

\csub{AI Generate}
\textbf{AI Generate} button $\to$ enter topic $\to$ optionally add source materials $\to$ Generate $\to$ preview/edit $\to$ \textbf{Create Draft Assignment}

\csub{AI Review}
Open an assignment $\to$ \textbf{AI Review} button $\to$ streaming review of wording, clarity, completeness

% ════════════════════════════════════════
\chead{Quizzes}
% ════════════════════════════════════════

\csub{Manual}
\textbf{+ New} $\to$ title, type $\to$ Create\\
\textbf{+ Add Question}:\\
\quad 1. Pick type (MC, T/F, Short, Essay, etc.)\\
\quad 2. Fill in question + answers\\
\quad 3. Add Question

\csub{AI Generate}
\textbf{AI Generate} $\to$ topic + sources + template $\to$ Generate $\to$ review/edit each Q $\to$ \textbf{Create Draft Quiz}

\csub{AI Review}
Load quiz $\to$ \textbf{AI Review} $\to$ identifies ambiguities, errors, trick questions

\csub{Question Types}
Multiple Choice, True/False, Short Answer, Essay, Numerical, Multiple Answers

% ════════════════════════════════════════
\chead{AI Assistant}
% ════════════════════════════════════════

\csub{Skills Panel (left)}
Clickable prompt macros for common tasks: generate quiz, generate exam, generate assignment, review quiz, review assignment, review exam.\\[2pt]
Click a skill $\to$ enter input $\to$ auto-sends to chat.

\csub{Chat (right)}
Free-form chat with Claude. Course context auto-included. Responses stream in real-time.

\chead{Prompt Templates}

\textbullet\ View/edit all prompt templates\\
\textbullet\ Placeholders: \texttt{\{topic\}}, \texttt{\{content\}}, \texttt{\{source\_material\}}, \texttt{\{num\_questions\}}, \texttt{\{question\_type\}}, \texttt{\{points\}}\\

\textbullet\ Changes saved to \texttt{prompts.json}

\chead{Key Workflows}
% ════════════════════════════════════════

\csub{AI Quiz from Readings}
Files tab: find reading $\to$\\
Quizzes: AI Generate $\to$\\
Add source (local file) $\to$\\
Set topic, \# questions $\to$\\
Generate $\to$ edit $\to$ Create Draft

\csub{Weekly Check-In}
Students tab: check missing/late $\to$\\
Calendar: see upcoming deadlines $\to$\\
Assignments: update descriptions

\csub{Review Before Publishing}
Load quiz or assignment $\to$\\
Click \textbf{AI Review} $\to$\\
Fix flagged issues $\to$ Publish
